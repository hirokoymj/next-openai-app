'use client';

import { action, getDropdownOptions } from './actions';
import { useActionState, useEffect, useState } from 'react';

const initialState = {
  success: false,
  firstName: '',
  lastName: '',
  food: '',
  gender: 'male',
  subjects: ['CS'],
};
interface Option {
  id: number;
  name: string;
}

export default function Home() {
  const [state, formAction, isPending] = useActionState(action, initialState);
  const [options, setOptions] = useState<Option[]>([]);
  const [resetKey, setResetKey] = useState(0);

  useEffect(() => {
    async function fetchOptions() {
      const data = await getDropdownOptions(); // call server action
      setOptions(data);
    }
    fetchOptions();
  }, []);

  const handleReset = () => {
    //const resetFormData = new FormData();
    //resetFormData.append('reset_form', 'true');

    // An async function with useActionState was called outside of a transition.
    // This is likely not what you intended (for example, isPending will not update correctly).
    // Either call the returned function inside startTransition, or pass it to an `action` or `formAction` prop.
    //formAction(resetFormData);
    setResetKey((prev) => prev + 1);
  };

  return (
    <div className="max-w-md mx-auto mt-10 p-6 bg-white rounded shadow-md">
      <h1 className="text-2xl font-bold mb-6 text-center">User Form</h1>

      <form action={formAction} key={resetKey} className="flex flex-col gap-4">
        {/* Username */}
        <div className="flex flex-col">
          <label htmlFor="firstName" className="mb-1 font-medium">
            First Name
          </label>
          <input
            id="firstName"
            type="text"
            name="firstName"
            className="border border-green-500 rounded px-3 py-2 focus:outline-none focus:ring-2 focus:ring-green-400"
          />
        </div>
        <div className="flex flex-col">
          <label htmlFor="lastName" className="mb-1 font-medium">
            Last Name
          </label>
          <input
            id="lastName"
            type="text"
            name="lastName"
            className="border border-green-500 rounded px-3 py-2 focus:outline-none focus:ring-2 focus:ring-green-400"
          />
        </div>

        {/* Dropdown */}
        <div className="flex flex-col">
          <label htmlFor="food" className="mb-1 font-medium">
            Favorite Food
          </label>
          <select
            id="food"
            name="food"
            className="border border-gray-300 rounded px-3 py-2 focus:outline-none focus:ring-2 focus:ring-blue-400">
            {options.map((option) => (
              <option key={option.id} value={option.name}>
                {option.name}
              </option>
            ))}
          </select>
        </div>

        {/* Gender Radio */}
        <div className="flex flex-col">
          <span className="mb-1 font-medium">Gender</span>
          <div className="flex gap-4">
            <label className="flex items-center gap-1">
              <input
                type="radio"
                name="gender"
                value="male"
                defaultChecked={state.gender === 'male'}
              />
              Male
            </label>
            <label className="flex items-center gap-1">
              <input
                type="radio"
                name="gender"
                value="female"
                defaultChecked={state.gender === 'female'}
              />
              Female
            </label>
          </div>
        </div>

        {/* Subjects Checkboxes */}
        <div className="flex flex-col">
          <span className="mb-1 font-medium">Subjects</span>
          <div className="flex flex-col gap-1">
            {['Math', 'English', 'CS'].map((subj) => (
              <label key={subj} className="flex items-center gap-2">
                <input
                  type="checkbox"
                  name="subject"
                  value={subj}
                  defaultChecked={state.subjects.includes(subj)}
                />
                {subj}
              </label>
            ))}
          </div>
        </div>

        {/* Submit Button */}
        <button
          type="submit"
          disabled={isPending}
          className="bg-blue-500 text-white py-2 rounded hover:bg-blue-600 transition disabled:opacity-50">
          {isPending ? 'Submitting...' : 'Submit'}
        </button>
        <button
          type="button"
          onClick={handleReset}
          disabled={isPending}
          className="bg-gray-300 text-black py-2 rounded hover:bg-gray-400 transition">
          Reset
        </button>

        {/* Server Response */}
        {state?.success && (
          <p className={state.success ? 'text-green-500' : 'text-red-500'}>
            {state.firstName}, {state.lastName}, {state.food}, {state.gender},{' '}
            {state.subjects.join(', ')}
          </p>
        )}
      </form>
    </div>
  );
}
